// Registro delle note attualmente accese su ciascun canale per evitare note bloccate let activeMidiNotes = new Map(); // key: `${channel}_${note}`, val: { channel, note, timeoutId } function sendMidiNoteOn(channel, note, velocity) { if (!document.getElementById('chkSendNotes').checked) return; if (!midiAccess || midiAccess.outputs.size === 0) return; const ch = Math.max(0, Math.min(15, (channel || 1) - 1)); const vel = Math.max(1, Math.min(127, velocity || 100)); const noteByte = note & 0x7F; const key = `${channel}_${note}`; // Se la nota era già attiva, rilasciala prima di risuonarla if (activeMidiNotes.has(key)) { clearTimeout(activeMidiNotes.get(key).timeoutId); sendMidiNoteOff(channel, note); } // Korg compatibile: Note-On (0x90 | ch, note, velocity) const raw = [0x90 | ch, noteByte, vel]; const selectedOut = document.getElementById('selMidiOut').value; if (selectedOut === 'ALL') { midiAccess.outputs.forEach(port => { try { port.send(raw); } catch (e) { try { port.send(new Uint8Array(raw)); } catch (err) {} } }); } else { const port = midiAccess.outputs.get(selectedOut); if (port) { try { port.send(raw); } catch (e) { try { port.send(new Uint8Array(raw)); } catch (err) {} } } } // Registra nota attiva activeMidiNotes.set(key, { channel, note, timeoutId: null }); flashMidiLed(); } function sendMidiNoteOff(channel, note) { if (!midiAccess || midiAccess.outputs.size === 0) return; const ch = Math.max(0, Math.min(15, (channel || 1) - 1)); const noteByte = note & 0x7F; const key = `${channel}_${note}`; // Rimuovi dal registro note attive activeMidiNotes.delete(key); // Doppia sicurezza: invia sia 0x80 con vel 0 sia Note-On con vel 0 (lo standard per NTS-1) const rawNoteOff = [0x80 | ch, noteByte, 0]; const rawNoteOnZero = [0x90 | ch, noteByte, 0]; const selectedOut = document.getElementById('selMidiOut').value; const dispatch = (port) => { try { port.send(rawNoteOnZero); port.send(rawNoteOff); } catch (e) { try { port.send(new Uint8Array(rawNoteOnZero)); port.send(new Uint8Array(rawNoteOff)); } catch (err) {} } }; if (selectedOut === 'ALL') { midiAccess.outputs.forEach(dispatch); } else { const port = midiAccess.outputs.get(selectedOut); if (port) dispatch(port); } } function sendMidiPanic() { // 1. Spegni singolarmente tutte le note che risultavano accese activeMidiNotes.forEach((val) => { if (val.timeoutId) clearTimeout(val.timeoutId); sendMidiNoteOff(val.channel, val.note); }); activeMidiNotes.clear(); if (!midiAccess) return; // 2. Invia CC 123 (All Notes Off), CC 120 (All Sound Off) e azzera pitch bend/mod su tutti i 16 canali midiAccess.outputs.forEach(port => { for (let ch = 0; ch < 16; ch++) { try { port.send([0xB0 | ch, 123, 0]); // All Notes Off port.send([0xB0 | ch, 120, 0]); // All Sound Off port.send([0xB0 | ch, 64, 0]); // Damper / Sustain Pedal Off // Note-off a tappeto su tutto il range dell'ottava centrale (da 36 a 84) for (let n = 36; n <= 84; n++) { port.send([0x90 | ch, n, 0]); } } catch (e) {} } }); showToast("MIDI PANIC: Note spente"); }
